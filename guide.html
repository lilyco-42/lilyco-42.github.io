🚀 webview-mini · 快速入门

方式 1: xmake (推荐)

-- xmake.lua
add_rules("mode.debug", "mode.release")
add_requires("webview-mini")

target("my-app")
    set_kind("binary")
    add_files("main.c")
    add_packages("webview-mini")
    add_syslinks("user32","shell32","ole32","oleaut32","shlwapi","version")
    set_languages("c11")
    if is_mode("release") then set_optimize("smallest") end
target_end()

然后将以下代码保存为 main.c:

#include "webview.h"
int main(void) {
    webview_t w = webview_create(0, NULL);
    webview_set_title(w, "MC Console");
    webview_set_size(w, 1100, 760, 0);
    webview_navigate(w, "http://192.168.10.165:8765");
    webview_run(w);
    webview_destroy(w);
    return 0;
}

编译运行:

xmake
xmake run

方式 2: 单头文件直链

下载 webview.h (211 KB),放到项目目录:

curl -O https://raw.githubusercontent.com/lilyco-42/webview-mini/main/webview.h
cl /O2 /MT main.c /link /SUBSYSTEM:WINDOWS user32.lib shell32.lib ole32.lib oleaut32.lib shlwapi.lib version.lib

🔌 webview-capi · 跨语言绑定

DLL 导出 17 个 C 函数,支持 PythonGoRustNode.js

Python (ctypes) · 5 行

import ctypes
lib = ctypes.CDLL('./webview.dll')
lib.webview_create.restype = ctypes.c_void_p
w = lib.webview_create(0, None)
lib.webview_set_title(w, b"Hello from Python")
lib.webview_navigate(w, b"https://example.com")
lib.webview_run(w)

Go (cgo) · 6 行

// #cgo LDFLAGS: -L. -lwebview
// #include "webview.h"
import "C"
w := C.webview_create(0, nil)
C.webview_set_title(w, C.CString("Hello from Go"))
C.webview_navigate(w, C.CString("https://example.com"))
C.webview_run(w)

📦 安装

  1. 下载 Release — webview-capi-v1.0.0-windows-x64.zip (150 KB)
  2. 解压到项目 — 得到 webview.dll + webview.h + webview.lib
  3. 复制示例代码 — 8 行 C 代码即可运行
  4. 编译链接 — cl /O2 /MT main.c /link webview.lib

⚠️ 避坑指南

Clash TUN 代理白屏

若 WebView2 窗口白屏(页面不加载),在代码中添加:

SetEnvironmentVariableW(L"WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS", L"--no-proxy-server");

DLL 导出失败

必须用 CMake 构建: cmake -DWEBVIEW_BUILD_SHARED_LIBRARY=ON ..

不要用 cl 直接编译 amalgamated webview.h —— 函数默认展开为 static inline。
