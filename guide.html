🚀 快速入门 (30 秒)

xmake (推荐)

-- xmake.lua
add_requires("webview-mini")
target("my-app")
    set_kind("binary")
    add_files("main.c")
    add_packages("webview-mini")
    add_syslinks("user32","shell32","ole32","shlwapi","version")
// main.c
#include "webview.h"
int main(void) {
    webview_t w = webview_create(0, NULL);
    webview_set_title(w, "MC Console");
    webview_set_size(w, 1100, 760, 0);
    webview_navigate(w, "http://192.168.10.165:8765");
    webview_run(w);
    webview_destroy(w);
    return 0;
}

xmake → 完成!

🔌 跨语言绑定 (C API DLL)

Python

import ctypes
lib = ctypes.CDLL('./webview.dll')
lib.webview_create.restype = ctypes.c_void_p
w = lib.webview_create(0, None)
lib.webview_set_title(w, b"Hello")
lib.webview_run(w)

Go

// #cgo LDFLAGS: -lwebview
// #include "webview.h"
w := C.webview_create(0, nil)
C.webview_run(w)

⚠️ 避坑

Clash TUN 代理白屏:

SetEnvironmentVariableW(L"WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS", L"--no-proxy-server");

DLL 导出失败: 必须用 cmake -DWEBVIEW_BUILD_SHARED_LIBRARY=ON 构建。
